# 构建网络
from pgmpy.models import BayesianNetwork
from pgmpy.factors.discrete import TabularCPD
from pgmpy.inference import VariableElimination

cancer_model = BayesianNetwork([('Pollution', 'Cancer'),
                                ('Smoker', 'Cancer'),
                                ('Cancer', 'Xray'),
                                ('Cancer', 'Dyspnoea')])
# 设置参数

poll = TabularCPD(variable='Pollution', variable_card=2,
                      values=[[0.9], [0.1]])
smoke = TabularCPD(variable='Smoker', variable_card=2,
                       values=[[0.3], [0.7]])
cancer = TabularCPD(variable='Cancer', variable_card=2,
                        values=[[0.03, 0.05, 0.001, 0.02],
                                [0.97, 0.95, 0.999, 0.98]],
                        evidence=['Smoker', 'Pollution'],
                        evidence_card=[2, 2])
xray = TabularCPD(variable='Xray', variable_card=2,
                      values=[[0.9, 0.2], [0.1, 0.8]],
                      evidence=['Cancer'], evidence_card=[2])
dysp = TabularCPD(variable='Dyspnoea', variable_card=2,
                      values=[[0.65, 0.3], [0.35, 0.7]],
                      evidence=['Cancer'], evidence_card=[2])
cancer_model.add_cpds(poll, smoke, cancer, xray, dysp)

# 测试网络结构是否正确
print(cancer_model.check_model())
# 变量消除法是精确推断的一种方法.

asia_infer = VariableElimination(cancer_model)
q = asia_infer.query(variables=['Cancer'], evidence={'Smoker': 0, 'Pollution': 0})
print(q)
