clc;
clear all;
clf;
% 加载 RGB.jpg 图片并转换为灰度图像
P1 = imread('模糊图1.jpg');
X1 = rgb2gray(P1); % 将彩色图像转换为灰度图像

% 加载 R-C.jpg 图片并转换为灰度图像
P2 = imread('模糊图2.jpg');
X2 = rgb2gray(P2); % 将彩色图像转换为灰度图像
%基于小波分解的图像融合
XFUS=wfusimg(X1,X2,'db4',2,'min','min')
subplot(1,3,1);imshow(X1);
axis square;
title('模糊图1');
subplot(1,3,2);imshow(X2);
axis square;
title('模糊图2');
subplot(1,3,3);imshow(uint8(XFUS));
axis square;
title('恢复后图像');
